import heapq

class Node:
    def __init__(self, state, parent=None, g=0, h=0):
        self.state = state
        self.parent = parent
        self.g = g
        self.h = h
        self.f = g + h
    
    def __lt__(self, other):
        return self.f < other.f

def a_star_search(start, goal, neighbors, heuristic):
    open_list = []
    closed_list = set()
    
    start_node = Node(start, g=0, h=heuristic(start, goal))
    goal_node = Node(goal)
    
    heapq.heappush(open_list, start_node)
    
    while open_list:
        current_node = heapq.heappop(open_list)
        
        if current_node.state == goal_node.state:
            path = []
            while current_node:
                path.append(current_node.state)
                current_node = current_node.parent
            return path[::-1]  # Return reversed path
        
        closed_list.add(current_node.state)
        
        for neighbor, cost in neighbors(current_node.state):
            if neighbor in closed_list:
                continue
            
            g = current_node.g + cost
            h = heuristic(neighbor, goal)
            neighbor_node = Node(neighbor, current_node, g, h)
            
            if any(node.state == neighbor and node.f <= neighbor_node.f for node in open_list):
                continue
            
            heapq.heappush(open_list, neighbor_node)
    
    return None  # No path found

# Example heuristic function (Manhattan distance for grid-based problems)
def manhattan_heuristic(node, goal):
    return abs(node[0] - goal[0]) + abs(node[1] - goal[1])

# Example neighbors function for a grid (returns neighbors and their costs)
def grid_neighbors(state):
    x, y = state
    neighbors = []
    if x > 0:
        neighbors.append(((x - 1, y), 1))
    if x < 4:  # Assuming grid width is 5
        neighbors.append(((x + 1, y), 1))
    if y > 0:
        neighbors.append(((x, y - 1), 1))
    if y < 4:  # Assuming grid height is 5
        neighbors.append(((x, y + 1), 1))
    return neighbors

# Example usage
start = (0, 0)
goal = (4, 4)
path = a_star_search(start, goal, grid_neighbors, manhattan_heuristic)
print("Path:", path)
